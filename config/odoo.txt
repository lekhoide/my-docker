# See all port has used
sudo netstat -tulnp

# Check volume use for container
docker ps -a --filter volume=<volume-name>

# Run dockerhub and log-in to container
docker run --rm -it <docker-hub> bash

# Different docker run and docker start
docker run: create new container from image, use when you new config to test
docker start: restart container exists has status Exited, use old config

# Check container when has issue and make Exited
docker logs <container>: see log in container
docker start <container> or docker compose start <container>
docker logs -f <container>

# access to shell container running
docker exec -it <container> bash

# create new container from image and access to shell
docker run --rm -it <image> bash

# See group folder and file
ls -l
